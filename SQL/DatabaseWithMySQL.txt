//INNER JOIN(dui table er just match roll gular details show korbe)
*SELECT sd.ROll,ex.Reg_Number,sd.Name,sd.Gender,ex.Group_Name,ex.GPA
FROM student_details AS sd INNER JOIN exam_result AS ex
ON sd.Roll=ex.Roll;
//LEFT JOIN(dui table er match roll gular details show korbe and left table er match charau baki details show korbe)
*SELECT sd.ROll,ex.Reg_Number,sd.Name,sd.Gender,ex.Group_Name,ex.GPA
FROM student_details AS sd LEFT JOIN exam_result AS ex
ON sd.Roll=ex.Roll;
//RIGHTT JOIN(dui table er match roll gular details show korbe and right table er match charau baki details show korbe)
*SELECT sd.ROll,ex.Reg_Number,sd.Name,sd.Gender,ex.Group_Name,ex.GPA
FROM student_details AS sd RIGHT JOIN exam_result AS ex
ON sd.Roll=ex.Roll;
